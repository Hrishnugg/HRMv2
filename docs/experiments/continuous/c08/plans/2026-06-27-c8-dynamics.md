```python
"""Deterministic moving circular obstacles + time-feasibility checks (C8 dynamics).

Pure geometry over numpy; no torch, no continuous_prm_common import needed beyond
optional typing. Agent is a point (MVP)."""
from __future__ import annotations

from dataclasses import dataclass
from typing import List, Sequence

import numpy as np


def _tri(x: float) -> float:
    """Triangle wave with period 1, range [0,1]: 0->1 over [0,0.5], 1->0 over [0.5,1]."""
    x = x - np.floor(x)
    return float(2.0 * x if x < 0.5 else 2.0 * (1.0 - x))


@dataclass
class MovingCircle:
    ax: float
    ay: float
    bx: float
    by: float
    period: float
    radius: float

    def center_at(self, t: float) -> np.ndarray:
        frac = _tri(t / self.period) if self.period > 0 else 0.0
        return np.array([self.ax + (self.bx - self.ax) * frac,
                         self.ay + (self.by - self.ay) * frac], dtype=np.float64)


class Dynamics:
    def __init__(self, circles: Sequence[MovingCircle]):
        self.circles: List[MovingCircle] = list(circles)

    def point_free(self, p: np.ndarray, t: float) -> bool:
        for c in self.circles:
            if np.linalg.norm(p - c.center_at(t)) <= c.radius:
                return False
        return True

    def node_free(self, p: np.ndarray, t0: float, t1: float, samples: int = 8) -> bool:
        ts = np.linspace(t0, t1, max(2, samples))
        return all(self.point_free(p, float(t)) for t in ts)

    def edge_free(self, a: np.ndarray, b: np.ndarray, t0: float, t1: float, samples: int = 16) -> bool:
        # Agent sweeps a->b linearly in time over [t0,t1]; check clearance vs each circle.
        ts = np.linspace(t0, t1, max(2, samples))
        for t in ts:
            frac = 0.0 if t1 <= t0 else (t - t0) / (t1 - t0)
            pos = a + (b - a) * frac
            if not self.point_free(pos, float(t)):
                return False
        return True
```

- [ ] **Step 3: implement** (`continuous_prm_spacetime.py`). Key points: edge step cost `ceil(L/(v*dt))`; move feasible iff `dyn.edge_free(points[u], points[v], t*dt, (t+steps)*dt)`; wait feasible iff `dyn.node_free(points[u], t*dt, (t+1)*dt)`; `g=(arrival time-step)`; priority `f = t + h[node, t]` (h in time-step units); closed set over `(node, t)`; `t_max` bounds time. Return `{"found","arrival","expansions","closed"}` (arrival in time-steps; `arrival=cost`).

```python
from __future__ import annotations
import heapq, math
from typing import Any, Dict, List, Tuple
import numpy as np


def _edge_steps(length: float, v_agent: float, dt: float) -> int:
    return max(1, int(math.ceil(length / (v_agent * dt))))


def space_time_astar_prm(adj, points, dyn, h_table, budget, v_agent, dt, t_max,
                         start=0, goal=1) -> Dict[str, Any]:
    n = len(adj)
    Tcap = int(h_table.shape[1]) - 1
    def hval(node, t):
        return float(h_table[node, min(t, Tcap)])
    g: Dict[Tuple[int, int], int] = {(start, 0): 0}
    counter = 0
    openq: List[Tuple[float, int, int, int]] = [(hval(start, 0), counter, start, 0)]
    closed = set()
    expansions = 0
    while openq and expansions < budget:
        f, _, u, t = heapq.heappop(openq)
        if (u, t) in closed:
            continue
        closed.add((u, t)); expansions += 1
        if u == goal:
            return {"found": True, "arrival": int(t), "expansions": expansions, "closed": len(closed)}
        # wait
        if t + 1 <= t_max and dyn.node_free(points[u], t * dt, (t + 1) * dt):
            nt = t + 1
            if (u, nt) not in closed and nt < g.get((u, nt), 1 << 30):
                g[(u, nt)] = nt; counter += 1
                heapq.heappush(openq, (nt + hval(u, nt), counter, u, nt))
        # moves
        for v, length in adj[u]:
            steps = _edge_steps(float(length), v_agent, dt)
            nt = t + steps
            if nt > t_max:
                continue
            if not dyn.edge_free(points[u], points[v], t * dt, nt * dt):
                continue
            if (v, nt) in closed:
                continue
            if nt < g.get((v, nt), 1 << 30):
                g[(v, nt)] = nt; counter += 1
                heapq.heappush(openq, (nt + hval(v, nt), counter, v, nt))
    return {"found": False, "arrival": -1, "expansions": expansions, "closed": len(closed)}


def backward_spacetime_dijkstra(adj, points, dyn, v_agent, dt, t_max, goal=1) -> np.ndarray:
    """h*[node, t] = min arrival time-step at goal starting from (node, t), or +inf.
    Computed by DP backward in time: h*[goal, t] = t; otherwise min over wait/move
    of the successor's h*. Iterate from t_max down to 0 until fixpoint (one sweep
    suffices because moves strictly increase t)."""
    n = len(adj)
    INF = float("inf")
    hstar = np.full((n, t_max + 1), INF, dtype=np.float64)
    # build reverse adjacency with the same step cost (length symmetric)
    for t in range(t_max, -1, -1):
        for u in range(n):
            if u == goal:
                hstar[u, t] = float(t)
                continue
            best = INF
            # wait
            if t + 1 <= t_max and dyn.node_free(points[u], t * dt, (t + 1) * dt):
                best = min(best, hstar[u, t + 1])
            # moves
            for v, length in adj[u]:
                steps = _edge_steps(float(length), v_agent, dt)
                nt = t + steps
                if nt <= t_max and dyn.edge_free(points[u], points[v], t * dt, nt * dt):
                    best = min(best, hstar[v, nt])
            hstar[u, t] = best
    return hstar
```

Note: `backward_spacetime_dijkstra` returns arrival-time-step values (so `h*[node,t] - t` = time-to-go). Convert to time-to-go where needed: `ttg = hstar - t`. For the Oracle *heuristic table* used by the search, pass `hstar - t_grid` (time-to-go), since the search expects `h[node,t]` = remaining time. Add a helper `oracle_time_to_go(hstar, t_max)` returning `hstar - arange(t_max+1)[None,:]` (clip negatives/inf to large finite). Add a test that `oracle_time_to_go` is 0 at the goal and finite where reachable.

```python
"""Time-aware heuristic providers returning h[node, t] tables (C8)."""
from __future__ import annotations
import abc
import numpy as np
import continuous_prm_spacetime as ST


def euclid_time_row(roadmap, v_agent, goal_idx=1) -> np.ndarray:
    goal = roadmap.points[goal_idx]
    return np.linalg.norm(roadmap.points - goal[None, :], axis=1) / float(v_agent)


def _finite_fill(a, fallback):
    out = np.array(a, dtype=np.float64); bad = ~np.isfinite(out)
    if bad.any(): out[bad] = fallback
    return np.maximum(out, 0.0)


class SpaceTimeHeuristicProvider(abc.ABC):
    name = "base"
    @abc.abstractmethod
    def h_table(self, world, roadmap, dyn, v_agent, dt, t_max, goal_idx=1) -> np.ndarray: ...


class EuclidTimeProvider(SpaceTimeHeuristicProvider):
    name = "euclid"
    def h_table(self, world, roadmap, dyn, v_agent, dt, t_max, goal_idx=1):
        row = euclid_time_row(roadmap, v_agent, goal_idx) / float(dt)  # in time-steps
        return _finite_fill(np.repeat(row[:, None], t_max + 1, axis=1), 0.0)


class OracleProvider(SpaceTimeHeuristicProvider):
    name = "oracle"
    def h_table(self, world, roadmap, dyn, v_agent, dt, t_max, goal_idx=1):
        hstar = ST.backward_spacetime_dijkstra(roadmap.adj, roadmap.points, dyn, v_agent, dt, t_max, goal_idx)
        ttg = ST.oracle_time_to_go(hstar, t_max)  # hstar - t, clipped
        fill = float(np.nanmax(ttg[np.isfinite(ttg)]) + t_max) if np.isfinite(ttg).any() else float(t_max * 2)
        return _finite_fill(ttg, fill)
```
